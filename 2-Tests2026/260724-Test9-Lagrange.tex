\item Using Lagrange multipliers, find the maximum and minimum values of the $f(x,y)=xy^2$ subject to the constraint $x^2 +2y^2 = 4$.

\begin{mysolution}
Let $g(x,y) = x^2 + 2y^2 = 4$. By the method of Lagrange multipliers, we set $\nabla f(x,y) = \lambda \nabla g(x,y)$:
\[
\begin{cases}
y^2 = 2\lambda x & \text{(1)} \\
2xy = 4\lambda y & \text{(2)} \\
x^2 + 2y^2 = 4 & \text{(3)}
\end{cases}
\]

From equation (2), we can rewrite it as $y(x - 2\lambda) = 0$, which yields two cases:

\begin{enumerate}[label=\roman*)]
    \item \textbf{Case $y = 0$:} \\
    Substituting $y = 0$ into the constraint (3) gives:
    \[
    x^2 + 2(0)^2 = 4 \implies x^2 = 4 \implies x = \pm 2.
    \]
    This provides the critical points $(2,0)$ and $(-2,0)$. Evaluating $f$ at these points:
    \[
    f(\pm 2, 0) = 0.
    \]

    \item \textbf{Case $y \neq 0$:} \\
    Since $y \neq 0$, we can divide equation (2) by $2y$ to find $x = 2\lambda$. Substituting $\lambda = \frac{x}{2}$ into equation (1) yields:
    \[
    y^2 = 2\left(\frac{x}{2}\right)x \implies y^2 = x^2.
    \]
    Substituting $y^2 = x^2$ into the constraint equation (3):
    \[
    x^2 + 2x^2 = 4 \implies 3x^2 = 4 \implies x = \pm \frac{2}{\sqrt{3}}.
    \]
    Since $y^2 = x^2 = \frac{4}{3}$, we get $y = \pm \frac{2}{\sqrt{3}}$. This produces four critical points:
    \[
    \left(\pm \frac{2}{\sqrt{3}}, \, \pm \frac{2}{\sqrt{3}}\right).
    \]
    Evaluating $f$ at these points:
    \[
    f\left(\frac{2}{\sqrt{3}}, \, \pm \frac{2}{\sqrt{3}}\right) = \left(\frac{2}{\sqrt{3}}\right)\left(\frac{4}{3}\right) = \frac{8}{3\sqrt{3}} = \frac{8\sqrt{3}}{9},
    \]
    \[
    f\left(-\frac{2}{\sqrt{3}}, \, \pm \frac{2}{\sqrt{3}}\right) = \left(-\frac{2}{\sqrt{3}}\right)\left(\frac{4}{3}\right) = -\frac{8}{3\sqrt{3}} = -\frac{8\sqrt{3}}{9}.
    \]
\end{enumerate}

Comparing all candidate values, we conclude:
\begin{itemize}
    \item The \textbf{maximum value} is $\displaystyle \frac{8\sqrt{3}}{9}$ (achieved at $\left(\frac{2}{\sqrt{3}}, \, \pm \frac{2}{\sqrt{3}}\right)$).
    \item The \textbf{minimum value} is $\displaystyle -\frac{8\sqrt{3}}{9}$ (achieved at $\left(-\frac{2}{\sqrt{3}}, \, \pm \frac{2}{\sqrt{3}}\right)$).
\end{itemize}
\end{mysolution}
